% ------------------------- key results -------------------------
\noindent{\coverlabel{Key results}}
\noindent{\sffamily\small\color{ink}\textbf{\color{navy}SoC}\enspace RVBL-2 core (RV32I + Zmmul + Xicrc) with GPIO (8 pins) and UART (115200 8-N-1), 25\,MHz}\par
\vspace{2.5mm}
\noindent\begin{tabular}{@{}*{2}{p{0.315\linewidth}@{\hspace{0.0275\linewidth}}}p{0.315\linewidth}@{}}
\coverstat[navy]{275}{instructions}{\code{main.s}, 1,100\,B, 8 commands; CRC-16 with \code{crcb}, 0.1\,°C with \code{mul}; latching EPO} &
\coverstat{PASS}{official test firmware}{on the ChipInventor platform: 20 checks, 0 failures; GPIO \hex{0xA} to \hex{0xA}, UART \hex{0x30} to \hex{0x30} (\secref{sec:official})} &
\coverstat{124\,/\,0}{checks\,/\,failures}{team testbench \code{tb\_data\_center\_application.v}, 12 cases, each also run on the platform (\secref{sec:validation})}\\[3mm]
\coverstat{12\,/\,12}{\code{./emu} runs verified}{each from reset, $\le$\,8 reply bytes, $\le$\,2.61\,ms of simulated time (\secref{sec:emu}, \secref{sec:validation})} &
\coverstat{4\,/\,4}{planted firmware bugs caught}{mutation test of the latching-EPO logic (\secref{sec:validation})} &
\coverstat{0 of 18,472}{bit-error patterns missed}{by CRC-16 on the 6-byte packet; an 8-bit XOR checksum misses 80 (\secref{sec:crc})}
\end{tabular}\par
\vspace{1.5mm}
\noindent{\sffamily\scriptsize\itshape\color{slate}Results from the ChipInventor platform and Icarus Verilog runs of the exported design; \secref{sec:validation} lists each test.}\par
