\newcommand{\cell}[5]{%
  \pgfmathsetmacro{\x}{\lw+(#2-1)*\cw}%
  \pgfmathsetmacro{\y}{-(#1-1)*\rh}%
  \node[anchor=north west, font=\sffamily\scriptsize, text=ink, inner sep=0pt]
       at (\x+1.1,\y-0.6) {#3};
  \node[anchor=north west, c#5, line width=0.55pt, minimum width=\cw mm-3.0mm, minimum height=3.1mm,
        inner sep=0.5pt, font=\ttfamily\scriptsize, text=ink] at (\x+1.1,\y-3.3) {#4};
  \ifnum#2>1 \draw[-{Stealth[length=2pt]}, hair, line width=0.5pt] (\x-0.9,\y-4.85) -- (\x+1.0,\y-4.85);\fi
}

  \node[anchor=north west, font=\sffamily\scriptsize, text=ink, inner sep=0pt, text width=\lw mm-8.5mm,
        align=left] at (7.4,\y-0.95) {#3};
}
